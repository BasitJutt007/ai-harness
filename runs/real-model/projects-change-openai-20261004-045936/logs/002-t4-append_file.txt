+      await createProject({ status: 'archived' }),
+      await createProject({ status: 'archived' }),
+      await createProject({ status: 'archived' }),
+    ];
+    await createProject({ status: 'active' });
+    await createProject({ status: 'active' });
+
+    const res = await request(app).get('/v1/projects').query({ status: 'archived', limit: '2' });

+    expect(firstPage.data).toHaveLength(2);
+    expect(firstPage.data.every((project) => project.status === 'archived')).toBe(true);
+    expect(firstPage.nextCursor).not.toBeNull();
+
+    const next = await request(app)
+      .get('/v1/projects')
+      .query({ status: 'archived', limit: '2', cursor: firstPage.nextCursor });
+    expect(next.status).toBe(200);
+    const secondPage = PageBody.parse(next.body);
+    expect(secondPage.data).toHaveLength(1);
+    expect(secondPage.data.every((project) => project.status === 'archived')).toBe(true);
+    expect(secondPage.nextCursor).toBeNull();
+
+    const ids = [...firstPage.data, ...secondPage.data].map((project) => project.id);
+    expect(new Set(ids)).toEqual(new Set(archived.map((project) => project.id)));
+  });
