+  it('filters by status and rejects unknown status', async () => {
+    const active = await createProject({ status: 'active' });
+    const archived = await createProject({ status: 'archived' });
+    const activePage = PageBody.parse((await request(app).get('/v1/projects').query({ status: 'active' })).body);

+  });
