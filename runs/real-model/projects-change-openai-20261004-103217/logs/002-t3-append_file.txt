+    const deleted = await request(app).delete(`/v1/projects/${project.id}`);

+    expectProblem(await request(app).get(`/v1/projects/${project.id}`), 404);
+  });

+  it('returns problem 404 for missing and repeated deletes', async () => {

+    const project = await createProject();
+    await request(app).delete(`/v1/projects/${project.id}`).expect(204);
+    expectProblem(await request(app).delete(`/v1/projects/${project.id}`), 404);
+  });

+  });
+
+  it('filters by active and archived status', async () => {
+    const active = await createProject({ status: 'active' });
+    const archived = await createProject({ status: 'archived' });
+    const activePage = PageBody.parse((await request(app).get('/v1/projects').query({ status: 'active' })).body);
+    const archivedPage = PageBody.parse((await request(app).get('/v1/projects').query({ status: 'archived' })).body);
+    expect(activePage.data.every((project) => project.status === 'active')).toBe(true);
+    expect(archivedPage.data.every((project) => project.status === 'archived')).toBe(true);
+    expect(activePage.data.map((project) => project.id)).toContain(active.id);
+    expect(archivedPage.data.map((project) => project.id)).toContain(archived.id);
+  });
+
+  it('paginates filtered results and follows the filtered cursor exactly once', async () => {

+    await createProject({ status: 'active' });
+    const ids = await listAll({ status: 'archived', limit: '1' });

+  });

+  it('returns all statuses without a filter and rejects unknown status', async () => {
