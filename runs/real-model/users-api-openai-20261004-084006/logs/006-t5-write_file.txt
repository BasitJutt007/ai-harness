+import { Router, type Request, type Response } from 'express';
+import { randomUUID } from 'node:crypto';
+import { z } from 'zod';
+import { conflict, notFound, unprocessable } from '../lib/problem.ts';
+import { CursorQuerySchema, paginate, pageSchema } from '../lib/pagination.ts';
+import { idempotency } from '../lib/idempotency.ts';
+
+const UserSchema = z.object({ id: z.uuid(), email: z.email(), name: z.string().min(1).max(100), role: z.enum(['admin', 'member']), createdAt: z.iso.datetime(), updatedAt: z.iso.datetime() });
+const CreateSchema = z.object({ email: z.email(), name: z.string().min(1).max(100), role: z.enum(['admin', 'member']).default('member') }).strict();
+const PatchSchema = CreateSchema.partial().strict();
+const IdSchema = z.object({ userId: z.uuid() });
+type User = z.infer<typeof UserSchema>;
+
+export function usersRouter(): Router {
+  const router = Router();
+  const users: User[] = [];
+  const find = (id: string): User => { const user = users.find((item) => item.id === id); if (!user) throw notFound(`User ${id} was not found.`); return user; };
+  router.post('/v1/users', idempotency(), (req: Request, res: Response) => {
+    const input = CreateSchema.parse(req.body);
+    if (users.some((u) => u.email === input.email)) throw conflict('email: already exists');
+    const now = new Date().toISOString();
+    const user = UserSchema.parse({ ...input, id: randomUUID(), createdAt: now, updatedAt: now });
+    users.push(user); res.location(`/v1/users/${user.id}`).status(201).json(UserSchema.parse(user));
+  });
+  router.get('/v1/users', (req: Request, res: Response) => { const query = CursorQuerySchema.parse(req.query); res.json(pageSchema(UserSchema).parse(paginate(users, query, (u) => u.id))); });
+  router.get('/v1/users/:userId', (req: Request, res: Response) => { const { userId } = IdSchema.parse(req.params); res.json(UserSchema.parse(find(userId))); });
+  router.patch('/v1/users/:userId', idempotency(), (req: Request, res: Response) => { const { userId } = IdSchema.parse(req.params); const user = find(userId); const input = PatchSchema.parse(req.body); if (Object.keys(input).length === 0) throw unprocessable('body: must not be empty'); if (input.email && users.some((u) => u.id !== userId && u.email === input.email)) throw conflict('email: already exists'); Object.assign(user, input, { updatedAt: new Date().toISOString() }); res.json(UserSchema.parse(user)); });
+  router.delete('/v1/users/:userId', (req: Request, res: Response) => { const { userId } = IdSchema.parse(req.params); find(userId); users.splice(users.findIndex((u) => u.id === userId), 1); res.status(204).send(); });
+  return router;
+}
