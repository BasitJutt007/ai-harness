+import request from 'supertest';
+import { describe, expect, it } from 'vitest';
+import { createApp } from '../src/app.ts';
+import { ProblemSchema } from '../src/lib/problem.ts';
+
+describe('users API', () => {
+  it('creates, gets, updates, lists with cursors, and deletes users', async () => {
+    const app = createApp();
+    const created = await request(app).post('/v1/users').send({ email: 'a@example.com', name: 'A' });
+    expect(created.status).toBe(201);
+    expect(created.headers.location).toMatch(/^\/v1\/users\//);
+    expect(created.body.role).toBe('member');
+    expect(created.body.email).toBe('a@example.com');
+    const id = created.body.id as string;
+    expect((await request(app).get(`/v1/users/${id}`)).status).toBe(200);
+    const updated = await request(app).patch(`/v1/users/${id}`).send({ name: 'AA' });
+    expect(updated.status).toBe(200);
+    expect(updated.body.name).toBe('AA');
+    const list = await request(app).get('/v1/users?limit=1');
+    expect(list.status).toBe(200);
+    expect(list.body.data).toHaveLength(1);
+    expect(list.body.nextCursor).toBeNull();
+    expect((await request(app).delete(`/v1/users/${id}`)).status).toBe(204);
+    expect((await request(app).get(`/v1/users/${id}`)).status).toBe(404);
+  });
+
+  it('validates inputs, uniqueness, missing users, and malformed JSON', async () => {
+    const app = createApp();
+    const invalid = await request(app).post('/v1/users').send({ email: 'bad', name: '' });
+    expect(invalid.status).toBe(422);
+    expect(ProblemSchema.parse(invalid.body).detail).toContain('email');
+    expect(ProblemSchema.parse(invalid.body).detail).toContain('name');
+    expect((await request(app).post('/v1/users').set('Content-Type', 'application/json').send('{')).status).toBe(400);
+    await request(app).post('/v1/users').send({ email: 'same@example.com', name: 'One' });
+    expect((await request(app).post('/v1/users').send({ email: 'same@example.com', name: 'Two' })).status).toBe(409);
+    expect((await request(app).get('/v1/users/not-a-uuid')).status).toBe(422);
+    expect((await request(app).patch('/v1/users/00000000-0000-0000-0000-000000000000').send({ name: 'x' })).status).toBe(404);
+    expect((await request(app).delete('/v1/users/00000000-0000-0000-0000-000000000000')).status).toBe(404);
+    expect((await request(app).patch('/v1/users/00000000-0000-0000-0000-000000000000').send({})).status).toBe(404);
+  });
+
+  it('replays POST idempotently and rejects changed reuse', async () => {
+    const app = createApp();
+    const first = await request(app).post('/v1/users').set('Idempotency-Key', 'u1').send({ email: 'i@example.com', name: 'I' });
+    const replay = await request(app).post('/v1/users').set('Idempotency-Key', 'u1').send({ email: 'i@example.com', name: 'I' });
+    expect(replay.status).toBe(201);
+    expect(replay.headers['idempotent-replayed']).toBe('true');
+    expect(replay.text).toBe(first.text);
+    expect((await request(app).post('/v1/users').set('Idempotency-Key', 'u1').send({ email: 'other@example.com', name: 'O' })).status).toBe(422);
+  });
+});
