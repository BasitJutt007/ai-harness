+import { randomUUID } from 'node:crypto';
+import type { Router } from 'express';
+import { z } from 'zod';
+import { idempotency } from '../lib/idempotency.ts';
+import { CursorQuerySchema, pageSchema, paginate } from '../lib/pagination.ts';
+import { conflict, notFound, unprocessable } from '../lib/problem.ts';
+
+const UserSchema = z.object({
+  id: z.uuid(),
+  email: z.email(),
+  name: z.string().min(1).max(100),
+  role: z.enum(['admin', 'member']),
+  createdAt: z.iso.datetime(),
+  updatedAt: z.iso.datetime(),
+});
+const CreateUserSchema = z.object({ email: z.email(), name: z.string().min(1).max(100), role: z.enum(['admin', 'member']).default('member') });
+const PatchUserSchema = z.object({ email: z.email().optional(), name: z.string().min(1).max(100).optional(), role: z.enum(['admin', 'member']).optional() }).refine((value) => Object.keys(value).length > 0, { message: 'at least one field is required' });
+const ParamsSchema = z.object({ userId: z.uuid() });
+const ListSchema = CursorQuerySchema;
+const UserPageSchema = pageSchema(UserSchema);
+
+type User = z.infer<typeof UserSchema>;
+
+export function registerUsers(router: Router): void {
+  const users = new Map<string, User>();
+  router.get('/v1/users', (req, res) => {
+    const query = ListSchema.parse(req.query);
+    const page = paginate([...users.values()], query, (user) => `${user.createdAt}|${user.id}`);
+    res.json(UserPageSchema.parse(page));
+  });
+  router.post('/v1/users', idempotency(), (req, res) => {
+    const body = CreateUserSchema.parse(req.body);
+    if ([...users.values()].some((user) => user.email === body.email)) throw conflict('email: Email already exists.');
+    const now = new Date().toISOString();
+    const user = UserSchema.parse({ id: randomUUID(), ...body, createdAt: now, updatedAt: now });
+    users.set(user.id, user);
+    res.status(201).location(`/v1/users/${user.id}`).json(UserSchema.parse(user));
+  });
+  router.get('/v1/users/:userId', (req, res) => {
+    const { userId } = ParamsSchema.parse(req.params);
+    const user = users.get(userId);
+    if (user === undefined) throw notFound(`user ${userId} not found`);
+    res.json(UserSchema.parse(user));
+  });
+  router.patch('/v1/users/:userId', idempotency(), (req, res) => {
+    const { userId } = ParamsSchema.parse(req.params);
+    const patch = PatchUserSchema.parse(req.body);
+    const user = users.get(userId);
+    if (user === undefined) throw notFound(`user ${userId} not found`);
+    if (patch.email !== undefined && [...users.values()].some((item) => item.id !== userId && item.email === patch.email)) throw conflict('email: Email already exists.');
+    const updated = UserSchema.parse({ ...user, ...patch, updatedAt: new Date().toISOString() });
+    users.set(userId, updated);
+    res.json(UserSchema.parse(updated));
+  });
+  router.delete('/v1/users/:userId', (req, res) => {
+    const { userId } = ParamsSchema.parse(req.params);
+    if (!users.delete(userId)) throw notFound(`user ${userId} not found`);
+    res.status(204).end();
+  });
+}
