+import { Router, type Request, type Response, type NextFunction } from 'express';
+import { z } from 'zod';
+import { randomUUID } from 'node:crypto';
+import { conflict, notFound, unprocessable } from '../lib/problem.ts';
+import { CursorQuerySchema, paginate, pageSchema } from '../lib/pagination.ts';
+import { idempotency } from '../lib/idempotency.ts';

+const userSchema = z.object({ id: z.uuid(), email: z.email(), name: z.string().min(1).max(100), role: z.enum(['admin', 'member']), createdAt: z.iso.datetime(), updatedAt: z.iso.datetime() });
+const createSchema = z.object({ email: z.email(), name: z.string().min(1).max(100), role: z.enum(['admin', 'member']).default('member') });
+const patchSchema = z.object({ email: z.email().optional(), name: z.string().min(1).max(100).optional(), role: z.enum(['admin', 'member']).optional() }).refine((value) => Object.keys(value).length > 0, 'at least one field is required');
+const idSchema = z.object({ userId: z.uuid() });
+const pageSchemaForUsers = pageSchema(userSchema);
+type User = z.infer<typeof userSchema>;
+type Store = { users: User[] };
+
+function parse<T>(schema: z.ZodType<T>, value: unknown): T {
+  const result = schema.safeParse(value);
+  if (!result.success) throw unprocessable(result.error.issues.map((issue) => `${issue.path.join('.') || 'request'}: ${issue.message}`).join('; '));
+  return result.data;
+}
+function forward(next: NextFunction, error: unknown): void { next(error); }
+
+export function createUsersRouter(): Router {
+  const store: Store = { users: [] };
+  const router = Router();
+  router.post('/', idempotency(), (req: Request, res: Response, next: NextFunction) => { try { const input = parse(createSchema, req.body); if (store.users.some((u) => u.email === input.email)) throw conflict('email already exists'); const now = new Date().toISOString(); const user: User = { id: randomUUID(), ...input, createdAt: now, updatedAt: now }; store.users.push(user); res.status(201).location(`/v1/users/${user.id}`).json(userSchema.parse(user)); } catch (error) { forward(next, error); } });
+  router.get('/', (req, res, next) => { try { const query = parse(CursorQuerySchema, req.query); res.json(pageSchemaForUsers.parse(paginate(store.users, query, (user) => user.id))); } catch (error) { forward(next, error); } });
+  router.get('/:userId', (req, res, next) => { try { const { userId } = parse(idSchema, req.params); const user = store.users.find((item) => item.id === userId); if (!user) throw notFound('user not found'); res.json(userSchema.parse(user)); } catch (error) { forward(next, error); } });
+  router.patch('/:userId', idempotency(), (req, res, next) => { try { const { userId } = parse(idSchema, req.params); const user = store.users.find((item) => item.id === userId); if (!user) throw notFound('user not found'); const input = parse(patchSchema, req.body); if (input.email !== undefined && store.users.some((item) => item.id !== userId && item.email === input.email)) throw conflict('email already exists'); Object.assign(user, input, { updatedAt: new Date().toISOString() }); res.json(userSchema.parse(user)); } catch (error) { forward(next, error); } });
+  router.delete('/:userId', (req, res, next) => { try { const { userId } = parse(idSchema, req.params); const index = store.users.findIndex((item) => item.id === userId); if (index < 0) throw notFound('user not found'); store.users.splice(index, 1); res.status(204).send(); } catch (error) { forward(next, error); } });
+  return router;
