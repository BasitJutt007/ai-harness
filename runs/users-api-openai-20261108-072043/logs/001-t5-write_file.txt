+import request from 'supertest';
+import { describe, expect, it } from 'vitest';
+import { createApp } from '../src/app.ts';
+
+const user = { email: 'a@example.com', name: 'Alice', role: 'admin' as const };
+
+describe('users API', () => {
+  it('creates, gets, updates and deletes users', async () => {
+    const app = createApp();
+    const created = await request(app).post('/v1/users').send(user);
+    expect(created.status).toBe(201);
+    expect(created.headers.location).toMatch(/^\/v1\/users\//);
+    expect(created.body).toMatchObject(user);
+    const id = created.body.id as string;
+    expect((await request(app).get(`/v1/users/${id}`)).status).toBe(200);
+    const patched = await request(app).patch(`/v1/users/${id}`).send({ name: 'A' });
+    expect(patched.status).toBe(200);
+    expect(patched.body.name).toBe('A');
+    expect((await request(app).delete(`/v1/users/${id}`)).status).toBe(204);
+    expect((await request(app).get(`/v1/users/${id}`)).status).toBe(404);
+  });
+
+  it('defaults role and cursor paginates all users', async () => {
+    const app = createApp();
+    for (let i = 0; i < 5; i += 1) await request(app).post('/v1/users').send({ email: `u${i}@e.com`, name: `U${i}` });
+    const ids: string[] = [];
+    let cursor: string | undefined;
+    do {
+      const response = await request(app).get('/v1/users').query({ limit: 2, ...(cursor ? { cursor } : {}) });
+      expect(response.status).toBe(200);
+      ids.push(...(response.body.data as { id: string }[]).map((item) => item.id));
+      cursor = response.body.nextCursor ?? undefined;
+    } while (cursor);
+    expect(ids).toHaveLength(5);
+    expect(new Set(ids).size).toBe(5);
+    expect((await request(app).get('/v1/users')).body.data[0].role).toBe('member');
+  });
+
+  it('validates fields, malformed JSON, missing resources and conflicts', async () => {
+    const app = createApp();
+    const invalid = await request(app).post('/v1/users').send({ email: 'bad', name: '' });
+    expect(invalid.status).toBe(422);
+    expect(invalid.headers['content-type']).toMatch(/application\/problem\+json/);
+    expect(invalid.body.detail).toMatch(/email/);
+    expect(invalid.body.detail).toMatch(/name/);
+    expect((await request(app).post('/v1/users').set('Content-Type', 'application/json').send('{')).status).toBe(400);
+    expect((await request(app).get('/v1/users').query({ limit: 101 })).status).toBe(422);
+    expect((await request(app).get('/v1/users/not-a-uuid')).status).toBe(422);
+    expect((await request(app).get('/v1/users/00000000-0000-4000-8000-000000000000')).status).toBe(404);
+    await request(app).post('/v1/users').send(user);
+    expect((await request(app).post('/v1/users').send(user)).status).toBe(409);
+    expect((await request(app).patch('/v1/users/00000000-0000-4000-8000-000000000000').send({ name: 'x' })).status).toBe(404);
+    expect((await request(app).patch('/v1/users/00000000-0000-4000-8000-000000000000').send({})).status).toBe(404);
+  });
+
+  it('replays idempotent creates and rejects changed bodies', async () => {
+    const app = createApp();
+    const first = await request(app).post('/v1/users').set('Idempotency-Key', 'key-1').send(user);
+    const second = await request(app).post('/v1/users').set('Idempotency-Key', 'key-1').send(user);
+    expect(second.status).toBe(201);
+    expect(second.headers['idempotent-replayed']).toBe('true');
+    expect(second.body.id).toBe(first.body.id);
+    expect((await request(app).post('/v1/users').set('Idempotency-Key', 'key-1').send({ ...user, name: 'Other' })).status).toBe(422);
+  });
+
+  it('rejects empty patches and duplicate email changes', async () => {
+    const app = createApp();
+    const a = await request(app).post('/v1/users').send(user);
+    const b = await request(app).post('/v1/users').send({ email: 'b@example.com', name: 'B' });
+    expect((await request(app).patch(`/v1/users/${a.body.id}`).send({})).status).toBe(422);
+    expect((await request(app).patch(`/v1/users/${b.body.id}`).send({ email: user.email })).status).toBe(409);
+  });
+});
