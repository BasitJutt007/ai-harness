+import express from 'express';
+import request from 'supertest';
+import { describe, expect, it } from 'vitest';
+import { createApp } from '../src/app.ts';
+
+const body = (email: string, name = 'Alice') => ({ email, name });
+const create = async (app: express.Express, value: object, key = `key-${Math.random()}`) => request(app).post('/v1/users').set('Idempotency-Key', key).send(value);
+
+describe('users API', () => {
+  it('creates with defaults and location, rejects duplicate email', async () => {
+    const app = createApp();
+    const response = await create(app, body('a@example.com'));
+    expect(response.status).toBe(201); expect(response.headers.location).toMatch(/^\/v1\/users\//);
+    expect(response.body.role).toBe('member'); expect(response.body.id).toEqual(expect.any(String));
+    const duplicate = await create(app, body('a@example.com'));
+    expect(duplicate.status).toBe(409); expect(duplicate.type).toBe('application/problem+json');
+  });
+  it('validates body, malformed JSON, and idempotency replay/reuse', async () => {
+    const app = createApp(); const key = 'same-key'; const value = body('b@example.com');
+    expect((await create(app, { email: 'bad', name: '' })).status).toBe(422);
+    const malformed = await request(app).post('/v1/users').set('Idempotency-Key', 'bad-json').set('Content-Type', 'application/json').send('{');
+    expect(malformed.status).toBe(400); expect(malformed.type).toBe('application/problem+json');
+    const first = await create(app, value, key); const replay = await create(app, value, key);
+    expect(replay.status).toBe(201); expect(replay.headers['idempotent-replayed']).toBe('true'); expect(replay.body.id).toBe(first.body.id);
+    expect((await create(app, body('other@example.com'), key)).status).toBe(422);
+  });
+  it('lists with cursor pagination and gets users', async () => {
+    const app = createApp(); await create(app, body('c1@example.com'), 'c1'); await create(app, body('c2@example.com'), 'c2');
+    const first = await request(app).get('/v1/users?limit=1'); expect(first.status).toBe(200); expect(first.body.data).toHaveLength(1); expect(first.body.nextCursor).toBeTruthy();
+    const second = await request(app).get(`/v1/users?limit=1&cursor=${encodeURIComponent(first.body.nextCursor)}`); expect(second.body.data).toHaveLength(1); expect(second.body.data[0].id).not.toBe(first.body.data[0].id);
+    expect((await request(app).get(`/v1/users/${first.body.data[0].id}`)).status).toBe(200);
+  });
+  it('patches partially, validates empty and duplicate email, and handles missing ids', async () => {
+    const app = createApp(); const made = await create(app, body('d@example.com'), 'd'); const id = made.body.id;
+    const patched = await request(app).patch(`/v1/users/${id}`).set('Idempotency-Key', 'patch-1').send({ name: 'Dora', role: 'admin' });
+    expect(patched.status).toBe(200); expect(patched.body.name).toBe('Dora'); expect(patched.body.email).toBe('d@example.com');
+    expect((await request(app).patch(`/v1/users/${id}`).set('Idempotency-Key', 'patch-2').send({})).status).toBe(422);
+    await create(app, body('e@example.com'), 'e'); expect((await request(app).patch(`/v1/users/${id}`).set('Idempotency-Key', 'patch-3').send({ email: 'e@example.com' })).status).toBe(409);
+    expect((await request(app).patch('/v1/users/00000000-0000-4000-8000-000000000000').set('Idempotency-Key', 'missing').send({ name: 'x' })).status).toBe(404);
+  });
+  it('deletes and reports missing resources and invalid parameters', async () => {
+    const app = createApp(); const made = await create(app, body('f@example.com'), 'f'); const id = made.body.id;
+    expect((await request(app).delete(`/v1/users/${id}`)).status).toBe(204); expect((await request(app).get(`/v1/users/${id}`)).status).toBe(404);
+    expect((await request(app).delete(`/v1/users/${id}`)).status).toBe(404); expect((await request(app).get('/v1/users/not-a-uuid')).status).toBe(422); expect((await request(app).get('/v1/users?limit=101')).status).toBe(422);
+  });
+});
