+describe('GET /v1/projects with status filter', () => {
+  it('returns only matching statuses and leaves the unfiltered list working', async () => {
+    const active = await createProject({ status: 'active' });
+    const archived = await createProject({ status: 'archived' });
+
+    const activeRes = await request(app).get('/v1/projects').query({ status: 'active', limit: '100' });
+    expect(activeRes.status).toBe(200);
+    const activePage = PageBody.parse(activeRes.body);
+    expect(activePage.data.every((project) => project.status === 'active')).toBe(true);
+    expect(activePage.data.map((project) => project.id)).toContain(active.id);
+    expect(activePage.data.map((project) => project.id)).not.toContain(archived.id);
+
+    const archivedRes = await request(app).get('/v1/projects').query({ status: 'archived', limit: '100' });
+    expect(archivedRes.status).toBe(200);
+    const archivedPage = PageBody.parse(archivedRes.body);
+    expect(archivedPage.data.every((project) => project.status === 'archived')).toBe(true);
+    expect(archivedPage.data.map((project) => project.id)).toContain(archived.id);
+    expect(archivedPage.data.map((project) => project.id)).not.toContain(active.id);
+
+    const allRes = await request(app).get('/v1/projects').query({ limit: '100' });
+    expect(allRes.status).toBe(200);
+    const allPage = PageBody.parse(allRes.body);
+    expect(allPage.data.map((project) => project.id)).toEqual(expect.arrayContaining([active.id, archived.id]));
+  });
+
+  it('paginates across the filtered set and rejects unknown statuses', async () => {
+    const archivedProjects = [
+      await createProject({ status: 'archived' }),
+      await createProject({ status: 'archived' }),
+      await createProject({ status: 'archived' }),
+    ];
+    await createProject({ status: 'active' });
+
+    const ids = await listAll({ status: 'archived', limit: '2' });

+
+    const firstPageRes = await request(app).get('/v1/projects').query({ status: 'archived', limit: '2' });
+    expect(firstPageRes.status).toBe(200);
+    const firstPage = PageBody.parse(firstPageRes.body);
