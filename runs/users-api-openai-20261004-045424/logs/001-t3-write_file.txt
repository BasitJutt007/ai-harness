+import request from 'supertest';
+import { describe, expect, it } from 'vitest';
+import { createApp } from '../src/app.ts';
+
+const baseUser = {
+  email: 'ada@example.com',
+  name: 'Ada Lovelace',
+};
+
+describe('users API', () => {
+  it('creates, gets, updates, lists, and deletes users', async () => {
+    const app = createApp();
+
+    const createResponse = await request(app)
+      .post('/v1/users')
+      .send(baseUser)
+      .expect(201);
+
+    expect(createResponse.headers.location).toMatch(/^\/v1\/users\/[0-9a-f-]{36}$/);
+    expect(createResponse.body).toMatchObject({
+      email: baseUser.email,
+      name: baseUser.name,
+      role: 'member',
+    });
+    expect(createResponse.body.id).toMatch(/[0-9a-f-]{36}/);
+    expect(createResponse.body.createdAt).toEqual(expect.any(String));
+    expect(createResponse.body.updatedAt).toEqual(expect.any(String));
+
+    const userId = createResponse.body.id as string;
+
+    const getResponse = await request(app).get(`/v1/users/${userId}`).expect(200);
+    expect(getResponse.body).toMatchObject(createResponse.body);
+
+    const patchResponse = await request(app)
+      .patch(`/v1/users/${userId}`)
+      .send({ name: 'Ada Byron', role: 'admin' })
+      .expect(200);
+    expect(patchResponse.body).toMatchObject({
+      id: userId,
+      email: baseUser.email,
+      name: 'Ada Byron',
+      role: 'admin',
+    });
+    expect(patchResponse.body.updatedAt).not.toEqual(createResponse.body.updatedAt);
+
+    const listResponse = await request(app).get('/v1/users').expect(200);
+    expect(listResponse.body.data).toHaveLength(1);
+    expect(listResponse.body.nextCursor).toBeNull();
+    expect(listResponse.body.data[0]).toMatchObject(patchResponse.body);
+
+    await request(app).delete(`/v1/users/${userId}`).expect(204);
+    await request(app).get(`/v1/users/${userId}`).expect(404);
+  });
+
+  it('returns problem json for duplicate email and missing users', async () => {
+    const app = createApp();
+
+    const first = await request(app).post('/v1/users').send({ ...baseUser, email: 'one@example.com' }).expect(201);
+    expect(first.body.email).toBe('one@example.com');
+
+    const duplicate = await request(app)
+      .post('/v1/users')
+      .send({ ...baseUser, email: 'one@example.com', name: 'Other' })
+      .expect(409);
+    expect(duplicate.headers['content-type']).toContain('application/problem+json');
+    expect(duplicate.body.status).toBe(409);
+
+    await request(app).get('/v1/users/00000000-0000-0000-0000-000000000000').expect(404);
+    await request(app).patch('/v1/users/00000000-0000-0000-0000-000000000000').send({ name: 'x' }).expect(404);
+    await request(app).delete('/v1/users/00000000-0000-0000-0000-000000000000').expect(404);
+  });
+
+  it('validates bodies, params, query, malformed json, and idempotency', async () => {
+    const app = createApp();
+
+    const invalidCreate = await request(app).post('/v1/users').send({ email: 'not-an-email', name: '' }).expect(422);
+    expect(invalidCreate.body.detail).toContain('email');
+    expect(invalidCreate.body.detail).toContain('name');
+
+    const malformed = await request(app)
+      .post('/v1/users')
+      .set('Content-Type', 'application/json')
+      .send('{');
+    expect(malformed.status).toBe(400);
+    expect(malformed.body.status).toBe(400);
+
+    const created = await request(app)
+      .post('/v1/users')
+      .set('Idempotency-Key', 'k1')
+      .send({ email: 'idem@example.com', name: 'Idem' })
+      .expect(201);
+    expect(created.headers['idempotent-replayed']).toBeUndefined();
+
+    const replayed = await request(app)
+      .post('/v1/users')
+      .set('Idempotency-Key', 'k1')
+      .send({ email: 'idem@example.com', name: 'Idem' })
+      .expect(201);
+    expect(replayed.headers['idempotent-replayed']).toBe('true');
+    expect(replayed.body).toEqual(created.body);
+
+    await request(app)
+      .post('/v1/users')
+      .set('Idempotency-Key', 'k1')
+      .send({ email: 'idem@example.com', name: 'Different' })
+      .expect(422);
+  });
+
+  it('paginates with cursor and limit', async () => {
+    const app = createApp();
+    const emails = Array.from({ length: 3 }, (_, index) => `user${index}@example.com`);
+    for (const email of emails) {
+      await request(app).post('/v1/users').send({ email, name: email }).expect(201);
+    }
+
+    const firstPage = await request(app).get('/v1/users?limit=2').expect(200);
+    expect(firstPage.body.data).toHaveLength(2);
+    expect(firstPage.body.nextCursor).toEqual(expect.any(String));
+
+    const secondPage = await request(app)
+      .get(`/v1/users?limit=2&cursor=${encodeURIComponent(firstPage.body.nextCursor)}`)
+      .expect(200);
+    expect(secondPage.body.data).toHaveLength(1);
+    expect(secondPage.body.nextCursor).toBeNull();
+
+    const seen = [...firstPage.body.data, ...secondPage.body.data].map((user) => user.email);
+    expect(seen).toEqual(emails);
+  });
+});
